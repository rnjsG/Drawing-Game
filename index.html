/* ====================================================== * Web Audio API 합성음 BGM 엔진 (수정본) * ====================================================== */ let audioCtx = null; let bgmTimer = null; let isBgmOff = false; let isMuted = false; let currentVolume = 0.3; const bgmMelody = [ 261.63, 329.63, 392.00, 523.25, 392.00, 329.63, 293.66, 349.23, 440.00, 587.33, 440.00, 349.23, 329.63, 392.00, 523.25, 659.25, 523.25, 392.00, 392.00, 440.00, 392.00, 349.23, 329.63, 293.66 ]; let melodyIdx = 0; // 클릭 이벤트가 발생한 순간에 오디오 컨텍스트 생성 및 시작 function ensureAudioContext() { if (!audioCtx) { const AudioContextClass = window.AudioContext || window.webkitAudioContext; audioCtx = new AudioContextClass(); } if (audioCtx.state === 'suspended') { audioCtx.resume(); } if (!bgmTimer && !isBgmOff) { startBGMSequence(); } } function playBGMNote() { if (!audioCtx || isBgmOff || isMuted) return; if (audioCtx.state === 'suspended') return; try { const osc = audioCtx.createOscillator(); const gain = audioCtx.createGain(); osc.type = 'sine'; osc.frequency.setValueAtTime(bgmMelody[melodyIdx], audioCtx.currentTime); gain.gain.setValueAtTime(currentVolume, audioCtx.currentTime); gain.gain.exponentialRampToValueAtTime(0.0001, audioCtx.currentTime + 0.22); osc.connect(gain); gain.connect(audioCtx.destination); osc.start(); osc.stop(audioCtx.currentTime + 0.23); melodyIdx = (melodyIdx + 1) % bgmMelody.length; } catch(e) { console.error(e); } } function startBGMSequence() { if (bgmTimer) clearInterval(bgmTimer); bgmTimer = setInterval(() => { playBGMNote(); }, 250); } // 화면 아무 곳이나 누르면 즉시 오디오 활성화 window.addEventListener('click', () => { ensureAudioContext(); }); function toggleBGM() { ensureAudioContext(); isBgmOff = !isBgmOff; document.getElementById('music-slash').style.display = isBgmOff ? 'block' : 'none'; if (isBgmOff) { if (bgmTimer) { clearInterval(bgmTimer); bgmTimer = null; } } else { startBGMSequence(); } } function toggleMute() { isMuted = !isMuted; document.getElementById('mute-icon').innerText = isMuted ? '🔇' : '🔊'; } function changeVolume(val) { currentVolume = parseFloat(val); }
